@app.get("/api/nerd/matches", tags=["nerd"])
async def get_nerd_matches(league: str = Query(None), db: AsyncSession = Depends(get_db)):
    try:
        base = """
            SELECT m.id, l.name as league, m.matchday,
                   m.match_datetime, th.name as home, ta.name as away,
                   m.home_goals, m.away_goals,
                   m."home_xG", m."away_xG",
                   m.home_ppda, m.away_ppda,
                   m.home_deep, m.away_deep,
                   m.home_xpts, m.away_xpts
            FROM matchcalendar m
            JOIN team th ON m.home_team_id = th.id
            JOIN team ta ON m.away_team_id = ta.id
            JOIN league l ON m.league_id = l.id
            WHERE m.is_completed = TRUE AND m.home_goals IS NOT NULL
        """
        params: dict = {}
        if league:
            base += " AND l.name ILIKE :league"
            params["league"] = f"%{league}%"
        base += " ORDER BY m.match_datetime DESC LIMIT 500"

        result = await db.execute(text(base), params)
        rows = result.fetchall()

        if rows:
            match_ids = [r[0] for r in rows]
            shots_res = await db.execute(
                text("""
                    SELECT match_id, team_type, COUNT(*) as cnt
                    FROM shots WHERE match_id = ANY(:ids)
                    GROUP BY match_id, team_type
                """),
                {"ids": match_ids},
            )
            shots_map: dict = {}
            for sr in shots_res.fetchall():
                mid, side, cnt = sr[0], normalize_team(sr[1]), sr[2]
                if mid not in shots_map:
                    shots_map[mid] = {"h": 0, "a": 0}
                shots_map[mid][side] = int(cnt)
        else:
            shots_map = {}

        return [
            {
                "id": r[0],
                "league": r[1],
                "matchday": r[2] or 0,
                "date": r[3].strftime("%Y-%m-%d %H:%M") if r[3] else "",
                "home": r[4],
                "away": r[5],
                "home_goals": r[6] or 0,
                "away_goals": r[7] or 0,
                "home_xG": sanitize_metric(r[8]),
                "away_xG": sanitize_metric(r[9]),
                "home_ppda": float(r[10]) if r[10] else 0.0,
                "away_ppda": float(r[11]) if r[11] else 0.0,
                "home_deep": int(r[12]) if r[12] else 0,
                "away_deep": int(r[13]) if r[13] else 0,
                "home_xpts": float(r[14]) if r[14] else 0.0,
                "away_xpts": float(r[15]) if r[15] else 0.0,
                "home_shots": shots_map.get(r[0], {}).get("h", 0),
                "away_shots": shots_map.get(r[0], {}).get("a", 0),
            }
            for r in rows
        ]
    except Exception as e:
        logger.error(f"nerd/matches error: {e}")
        return []


@app.get("/api/nerd/matches/{match_id}/rosters", tags=["nerd"])
async def get_nerd_rosters(match_id: int, db: AsyncSession = Depends(get_db)):
    try:
        result = await db.execute(
            text("""
                SELECT player, position, team_type, "time", goals, assists,
                       shots, key_passes, "xG", "xA"
                FROM rosters
                WHERE match_id = :mid
                ORDER BY team_type, "time" DESC
            """),
            {"mid": match_id},
        )
        rows = result.fetchall()
        home, away = [], []
        for r in rows:
            entry = {
                "player": r[0],
                "position": r[1] or "",
                "time": int(r[3]) if r[3] else 0,
                "goals": int(r[4]) if r[4] else 0,
                "assists": int(r[5]) if r[5] else 0,
                "shots": int(r[6]) if r[6] else 0,
                "key_passes": int(r[7]) if r[7] else 0,
                "xG": sanitize_metric(r[8]),
                "xA": sanitize_metric(r[9]),
            }
            if normalize_team(r[2]) == "h":
                home.append(entry)
            else:
                away.append(entry)
        return {"home": home, "away": away}
    except Exception as e:
        logger.error(f"nerd/rosters {match_id} error: {e}")
        return {"home": [], "away": []}

function MatchDetail({ matchId, homeTeam, awayTeam }: { matchId: number; homeTeam: string; awayTeam: string }) {
  const [shots, setShots] = useState<Shot[]>([]);
  const [rosters, setRosters] = useState<{ home: RosterEntry[]; away: RosterEntry[] }>({ home: [], away: [] });
  const [loading, setLoading] = useState(true);
  const [subTab, setSubTab] = useState<"rosters" | "shots">("rosters");

  useEffect(() => {
    Promise.all([
      fetch(`http://localhost:8000/api/matches/${matchId}/details`).then(r => r.json()),
      fetch(`http://localhost:8000/api/nerd/matches/${matchId}/rosters`).then(r => r.json()),
    ]).then(([details, ros]) => {
      setShots(details.shots || []);
      setRosters(ros);
    }).catch(console.error).finally(() => setLoading(false));
  }, [matchId]);

  if (loading) return (
    <div className="py-6 text-center text-[#8b949e] text-xs font-mono animate-pulse">

  { key: "shots", label: "Sh" },
  { key: "key_passes", label: "KP" },
  { key: "xG", label: "xG" },
  { key: "xA", label: "xA" },
  { key: "xGChain", label: "xGChain" },
  { key: "xGBuildup", label: "xGBuildup" },
] as const;

type ColKey = (typeof ALL_COLS)[number]["key"];

const DEFAULT_COLS: ColKey[] = ["time", "goals", "assists", "shots", "key_passes", "xG", "xA"];
const fmt = (n: number, d = 2) => Number(n).toFixed(d);

export default function PlayersTab() {
  const [query, setQuery] = useState("");
  const [suggestions, setSuggestions] = useState<string[]>([]);
  const [selected, setSelected] = useState<string | null>(null);
  const [totals, setTotals] = useState<PlayerTotals | null>(null);
  const [matches, setMatches] = useState<PlayerMatch[]>([]);
  const [shots, setShots] = useState<PlayerShot[]>([]);
  const [loading, setLoading] = useState(false);
  const [showShots, setShowShots] = useState(false);
  const [cols, setCols] = useState<Set<ColKey>>(new Set(DEFAULT_COLS));
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    if (query.length < 2) { setSuggestions([]); return; }
    if (debounceRef.current) clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(() => {
      fetch(`http://localhost:8000/api/scout/search?q=${encodeURIComponent(query)}`)
        .then(r => r.json()).then(setSuggestions).catch(() => setSuggestions([]));
    }, 300);
  }, [query]);

  const selectPlayer = async (name: string) => {
    setSelected(name);
    setQuery(name);
    setSuggestions([]);
    setLoading(true);
    try {
      const [profile, perMatch] = await Promise.all([
        fetch(`http://localhost:8000/api/scout/player/${encodeURIComponent(name)}`).then(r => r.json()),
        fetch(`http://localhost:8000/api/nerd/player/${encodeURIComponent(name)}/matches`).then(r => r.json()),
      ]);
      setTotals(profile.totals || null);
      setShots(profile.shots_map || []);
      setMatches(Array.isArray(perMatch) ? perMatch : []);
    } catch (e) {
      console.error(e);
    } finally {
      setLoading(false);
    }
  };

  const toggleCol = (k: ColKey) => {
    setCols(prev => { const n = new Set(prev); n.has(k) ? n.delete(k) : n.add(k); return n; });
  };

  const activeCols = ALL_COLS.filter(c => cols.has(c.key));

  const pillData = useMemo(() => totals ? [
    { label: "MIN",      value: totals.min,        color: "text-[#8b949e]" },
    { label: "GOL",      value: totals.g,           color: "text-[#3fb950]" },
    { label: "ASSIST",   value: totals.a,           color: "text-[#58a6ff]" },
    { label: "xG",       value: fmt(totals.xg),     color: "text-[#f0883e]" },
    { label: "xA",       value: fmt(totals.xa),     color: "text-[#d2a8ff]" },
    { label: "TIRI",     value: totals.shots,       color: "text-[#e6edf3]" },
    { label: "KP",       value: totals.key_passes,  color: "text-[#e6edf3]" },
    { label: "xGChain",  value: fmt(totals.xGChain),   color: "text-[#79c0ff]" },
    { label: "xGBuild",  value: fmt(totals.xGBuildup), color: "text-[#79c0ff]" },
  ] : [], [totals]);

  return (
    <div>
      {/* Search */}
      <div className="relative mb-6 max-w-md">
        <input
          value={query}
          onChange={e => { setQuery(e.target.value); setSelected(null); }}
          placeholder="Cerca giocatore..."
          className="w-full bg-[#161b22] border border-[#30363d] text-[#e6edf3] text-sm font-mono px-4 py-2.5 rounded focus:outline-none focus:border-[#58a6ff] placeholder-[#8b949e]"
        />
        {suggestions.length > 0 && (
          <div className="absolute top-full left-0 right-0 z-10 bg-[#161b22] border border-[#30363d] rounded mt-1 shadow-2xl overflow-hidden">
            {suggestions.map(s => (
              <button key={s} onClick={() => selectPlayer(s)}
                className="w-full text-left px-4 py-2 text-xs font-mono text-[#e6edf3] hover:bg-[#1c2128] transition-colors">
                {s}
              </button>
            ))}
          </div>
        )}
      </div>

      {loading && (
        <div className="text-center py-8 text-[#8b949e] text-xs font-mono animate-pulse">Caricamento...</div>
      )}

      {!loading && !selected && (
        <div className="text-center py-20 text-[#8b949e]">
          <div className="text-4xl mb-3 font-mono text-[#30363d]">_</div>
          <p className="text-xs font-mono">Cerca un giocatore per vederne le statistiche</p>
        </div>
      )}

      {totals && !loading && (
        <>
          {/* Totals pills */}
          <div className="flex flex-wrap gap-2 mb-6">
            {pillData.map(p => (
              <div key={p.label} className="bg-[#161b22] border border-[#30363d] rounded px-3 py-2 text-center min-w-[60px]">
                <div className="text-[9px] text-[#8b949e] font-mono uppercase tracking-widest mb-0.5">{p.label}</div>
                <div className={`text-base font-mono font-bold ${p.color}`}>{p.value}</div>
              </div>
            ))}
          </div>

          {/* Column toggles */}
          <div className="flex flex-wrap gap-1.5 mb-3 items-center">
            <span className="text-[9px] font-mono text-[#8b949e] uppercase tracking-widest mr-1">Colonne:</span>
            {ALL_COLS.map(c => (
              <button key={c.key} onClick={() => toggleCol(c.key)}
                className={`text-[10px] font-mono px-2 py-1 rounded border transition-all ${cols.has(c.key) ? "bg-[#1f6feb33] border-[#58a6ff] text-[#58a6ff]" : "bg-transparent border-[#30363d] text-[#8b949e] hover:border-[#8b949e]"}`}>
                {c.label}
              </button>
            ))}
          </div>

          {/* Per-match table */}
          <div className="overflow-x-auto mb-6 border border-[#30363d] rounded">
            <table className="w-full text-xs font-mono">
              <thead>
                <tr className="text-[#8b949e] border-b border-[#30363d] bg-[#161b22] text-[10px] uppercase tracking-wider">
                  <th className="text-left py-2 px-3">Data</th>
                  <th className="text-left py-2 px-3">Partita</th>
                  {activeCols.map(c => (
                    <th key={c.key} className="text-right py-2 px-2">{c.label}</th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {matches.map((m, i) => (
                  <tr key={i} className="border-b border-[#21262d] hover:bg-[#161b22] transition-colors">
                    <td className="py-1.5 px-3 text-[#8b949e] whitespace-nowrap">{m.date}</td>
                    <td className="py-1.5 px-3 whitespace-nowrap">
                      <span className={m.side === "H" ? "text-[#58a6ff] font-bold" : "text-[#8b949e]"}>{m.home}</span>
                      <span className="text-[#8b949e] mx-1.5">vs</span>
                      <span className={m.side === "A" ? "text-[#58a6ff] font-bold" : "text-[#8b949e]"}>{m.away}</span>
                    </td>
                    {activeCols.map(c => {
                      const raw = m[c.key as keyof PlayerMatch];
                      const num = typeof raw === "number" ? raw : 0;
                      let color = "text-[#e6edf3]";
                      if (c.key === "goals" && num > 0) color = "text-[#3fb950] font-bold";
                      if (c.key === "assists" && num > 0) color = "text-[#58a6ff] font-bold";
                      if (c.key === "xG" || c.key === "xA") color = "text-[#f0883e]";
                      if (c.key === "xGChain" || c.key === "xGBuildup") color = "text-[#79c0ff]";
                      if (c.key === "time") color = "text-[#8b949e]";
                      if (c.key === "position") color = "text-[#8b949e]";
                      const display = c.key === "time"
                        ? `${num}'`
                        : c.key === "position"
                        ? String(raw || "")
                        : ["xG", "xA", "xGChain", "xGBuildup"].includes(c.key)
                        ? fmt(num)
                        : (["goals", "assists"].includes(c.key) && num === 0) ? "" : String(raw ?? "");
                      return (
                        <td key={c.key} className={`py-1.5 text-right px-2 ${color}`}>{display}</td>
                      );
                    })}
                  </tr>
                ))}
                {matches.length === 0 && (
                  <tr>
                    <td colSpan={2 + activeCols.length} className="py-8 text-center text-[#8b949e] text-xs font-mono">
                      Nessuna partita trovata
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>

          {/* Shots section */}
          <button
            onClick={() => setShowShots(v => !v)}
            className="text-xs font-mono text-[#58a6ff] hover:text-[#79c0ff] mb-3 flex items-center gap-1 transition-colors"
          >
            <span className="font-bold">{showShots ? "▼" : "▶"}</span>
            Tabella tiri ({shots.length})
          </button>

          {showShots && (
            <div className="overflow-x-auto border border-[#30363d] rounded">
              <table className="w-full text-xs font-mono">
                <thead>
                  <tr className="text-[#8b949e] border-b border-[#30363d] bg-[#161b22] text-[10px] uppercase tracking-wider">
                    <th className="text-left py-2 px-3">Min</th>
                    <th className="text-right py-2 px-3">xG</th>
                    <th className="text-left py-2 px-3">Risultato</th>
                    <th className="text-left py-2 px-3">Situazione</th>
                    <th className="text-left py-2 px-3">Tipo</th>
                  </tr>
                </thead>
                <tbody>
                  {shots.map((s, i) => (
                    <tr key={i} className={`border-b border-[#21262d] hover:bg-[#161b22] ${s.result === "Goal" ? "bg-[#3fb95015]" : ""}`}>
                      <td className="py-1.5 px-3 text-[#8b949e]">{s.minute}&apos;</td>
                      <td className="py-1.5 px-3 text-right text-[#f0883e]">{fmt(s.xG)}</td>
                      <td className={`py-1.5 px-3 font-bold ${s.result === "Goal" ? "text-[#3fb950]" : s.result === "SavedShot" ? "text-[#58a6ff]" : "text-[#8b949e]"}`}>{s.result}</td>
                      <td className="py-1.5 px-3 text-[#8b949e]">{s.situation}</td>
                      <td className="py-1.5 px-3 text-[#8b949e]">{s.shotType}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </>
      )}
    </div>
  );
}
